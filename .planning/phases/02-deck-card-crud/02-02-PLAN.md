# Deck 모델 (이미 존재 — Phase 1)
class Deck < ApplicationRecord
  enum :scope, { official: 0, club: 1, personal: 2 }
  belongs_to :owner, polymorphic: true, optional: true   # official 은 owner nil
  has_many :cards, dependent: :destroy
  validates :title, presence: true
end

# DeckPolicy (Phase 1 stub — 본 plan에서 수정하지 않음)
class DeckPolicy < ApplicationPolicy
  def show?    = personal_owner?     # Phase 4가 official/club 확장 예정
  def update?  = personal_owner?
  def destroy? = personal_owner?
  def create?  = user.present?       # Phase 4가 scope별 분기 예정

  private
  def personal_owner?
    record.respond_to?(:owner) && record.owner == user
  end

  class Scope < Scope
    def resolve = scope.none           # 절대 호출 금지 (scope.none 반환)
  end
end

    ```ruby
    class DecksController < ApplicationController
      allow_unauthenticated_access only: %i[index show]
      before_action :resume_session, only: %i[index show]
      before_action :set_deck, only: %i[show edit update destroy]
      before_action :authorize_deck!, only: %i[edit update destroy]

      def index
        @official_decks = Deck.where(scope: :official).includes(:cards).order(:created_at)

        if authenticated?
          @personal_decks = Deck.where(scope: :personal, owner: Current.user)
                                .includes(:cards)
                                .order(updated_at: :desc)
          approved_club_ids = Current.user.club_memberships.approved.pluck(:club_id)
          @show_club_section = approved_club_ids.any?
          @club_decks = @show_club_section ?
            Deck.where(scope: :club, owner_type: "Club", owner_id: approved_club_ids)
                .includes(:cards).order(updated_at: :desc) : []
        else
          @personal_decks = []
          @club_decks = []
          @show_club_section = false
        end
      end

      def show
        # personal 덱이면 personal_owner? 통과해야 함; official 덱은 누구나 read.
        authorize @deck if @deck.personal?
        # @cards는 Wave 2가 사용. Wave 1에서는 정의만.
        @cards = @deck.cards.order(:position, :id)
      end

      def new
        @deck = Deck.new(scope: :personal, owner: Current.user)
        authorize @deck
      end

      def create
        @deck = Deck.new(deck_params.merge(scope: :personal, owner: Current.user))
        authorize @deck
        if @deck.save
          redirect_to @deck, notice: "덱이 생성되었습니다."
        else
          render :new, status: :unprocessable_entity
        end
      end

      def edit; end

      def update
        if @deck.update(deck_params)
          redirect_to @deck, notice: "덱이 수정되었습니다."
        else
          render :edit, status: :unprocessable_entity
        end
      end

      def destroy
        @deck.destroy!
        redirect_to decks_path, notice: "덱이 삭제되었습니다."
      end

      private

      def set_deck
        @deck = Deck.find(params[:id])
      end

      def authorize_deck!
        authorize @deck
      end

      def deck_params
        params.require(:deck).permit(:title, :description)
        # owner/scope/owner_id/owner_type 명시 제외 — T-02-01 (mass assignment 차단)
      end
    end
    ```

    `ApplicationController`가 Pundit::Authorization include + rescue_from Pundit::NotAuthorizedError 처리하므로 NotAuthorizedError가 알아서 403로 리턴됨.
