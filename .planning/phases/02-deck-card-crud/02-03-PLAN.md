  has_one_attached :front_image                              # 모델 attach 존재, UI 차단

class CardPolicy < ApplicationPolicy
  def create?  = deck_owner?
  def update?  = deck_owner?
  def destroy? = deck_owner?

  private
  def deck_owner?
    record.respond_to?(:deck) && record.deck&.owner == user
  end
end

    ```ruby
    class CardsController < ApplicationController
      before_action :set_deck
      before_action :set_card, only: %i[edit update destroy]

      def create
        @card = @deck.cards.build(card_params.merge(card_type: :basic))
        authorize @card
        if @card.save
          @cards = @deck.cards.order(:position, :id)
          respond_to do |format|
            format.turbo_stream
            format.html { redirect_to deck_path(@deck), notice: "카드가 추가되었습니다." }
          end
        else
          respond_to do |format|
            format.turbo_stream {
              render turbo_stream: turbo_stream.replace(
                "deck-add-form-#{@deck.id}",
                partial: "decks/card_add_form",
                locals: { deck: @deck, card: @card }
              ), status: :unprocessable_entity
            }
            format.html { render "decks/show", status: :unprocessable_entity }
          end
        end
      end

      def edit
        authorize @card
      end

      def update
        authorize @card
        if @card.update(card_params)
          respond_to do |format|
            format.turbo_stream
            format.html { redirect_to deck_path(@deck) }
          end
        else
          render turbo_stream: turbo_stream.replace(
            "card_#{@card.id}",
            partial: "decks/card_edit_form",
            locals: { card: @card, deck: @deck }
          ), status: :unprocessable_entity
        end
      end

      def destroy
        authorize @card
        @card.destroy!
        respond_to do |format|
          format.turbo_stream
          format.html { redirect_to deck_path(@deck) }
        end
      end

      private

      def set_deck
        # policy_scope(Deck) 사용 금지 (scope.none) — 명시적 owner 스코프
        @deck = Deck.where(owner: Current.user).find(params[:deck_id])
      end

      def set_card
        @card = @deck.cards.find(params[:id])
      end

      def card_params
        params.require(:card).permit(:front, :back, :hint, :tags)
        # 명시적 제외:
        #   :front_image, :back_image    — T-02-03 (CONTEXT.md Decision #6)
        #   :card_type                   — 컨트롤러에서 :basic 강제 (v1)
        #   :deck_id, :position          — 경로/모델이 결정
        #   :owner_id, :owner_type       — Deck의 책임
      end
    end
    ```

    **card_form_controller.js** (PATTERNS.md `card_form_controller.js` 섹션 + Decision #1, #4):
    ```javascript
    import { Controller } from "@hotwired/stimulus"

    export default class extends Controller {
      static targets = ["front", "counter"]
      static values = { savedCount: { type: Number, default: 0 } }

      handleKeydown(event) {
        // Cmd/Ctrl+Enter: 저장하고 다음 (Decision #1)
        if ((event.metaKey || event.ctrlKey) && event.key === "Enter") {
          event.preventDefault()
          // form.submit() 사용 금지 (Turbo 인터셉트 불가, Pitfall 5)
          this.element.querySelector("form").requestSubmit()
          return
        }
        // Escape: 인라인 편집 frame이면 read-only 행으로 복귀 (Decision #4)
        if (event.key === "Escape") {
          const frame = this.element.closest("turbo-frame")
          if (frame && frame.id.startsWith("card_")) {
            // 인라인 편집 컨텍스트: frame의 src를 비워 원본 행 복원
            // (Turbo가 자동으로 frame을 read-only partial로 재요청)
            event.preventDefault()
            frame.removeAttribute("src")
            frame.reload?.()  // Turbo 8+ frame.reload(); fallback: location.reload()
          }
          // add form은 frame 밖에 위치 — Escape 무시 (no-op)
        }
      }

      // Decision #1: 서버가 카운터를 렌더하므로 클라이언트는 value 변경 시 시각적 강조만
      savedCountValueChanged(newValue, oldValue) {
        if (this.hasCounterTarget && oldValue !== undefined && newValue > oldValue) {
          this.counterTarget.classList.add("card-form__counter--pulse")
          setTimeout(() => this.counterTarget.classList.remove("card-form__counter--pulse"), 300)
        }
      }
    }
    ```
    - **인라인 편집 (`_card_edit_form.html.erb`):** 편집 form은 `cards/edit.html.erb`의 `turbo_frame_tag "card_#{@card.id}"` 안에 렌더됨. `<td colspan="5">`에 `data-controller="card-form"` + `data-action="keydown->card-form#handleKeydown"` 적용 (본 plan의 `_card_edit_form.html.erb` ERB 블록 참조) → Escape 키 → `handleKeydown`이 `closest("turbo-frame")` 으로 부모 frame을 찾아 `frame.reload()` 호출 → Turbo가 GET `edit_deck_card_path`를 다시 요청하는 대신 frame의 src를 비우면 부모 페이지 컨텍스트(deck show)에서 해당 frame이 원본 read-only `_card_row.html.erb`로 자연 복원됨.
